from settings import _parse_yaml, _load_env_file, CONFIG_DIR
import os

# Proposed fix: defaults FIRST, then yaml, then env vars
def load_settings_fixed(env=None):
    env = env or os.environ.get('APP_ENV', 'production')
    settings = {}
    for key, value in _load_env_file(CONFIG_DIR + '/defaults.env').items():
        settings[key.lower()] = value
    settings.update(_parse_yaml(CONFIG_DIR + '/' + env + '.yaml'))
    for key in ('DATABASE_URL','REDIS_URL','LOG_LEVEL'):
        if key in os.environ:
            settings[key.lower()] = os.environ[key]
    return settings

print('--- staging with fix ---')
print(load_settings_fixed('staging'))
print('--- production with fix ---')
print(load_settings_fixed('production'))
print('--- no APP_ENV (bare container) with fix ---')
print(load_settings_fixed(None))
