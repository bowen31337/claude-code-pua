import re, pathlib
p = pathlib.Path("settings.py")
src = p.read_text()

old = '''    settings = {}

    # 1. environment-specific yaml
    settings.update(_parse_yaml(os.path.join(CONFIG_DIR, f"{env}.yaml")))

    # 2. baked-in defaults, so a bare container still boots
    for key, value in _load_env_file(os.path.join(CONFIG_DIR, "defaults.env")).items():
        settings[key.lower()] = value

    # 3. real environment variables win
    for key in ("DATABASE_URL", "REDIS_URL", "LOG_LEVEL"):
        if key in os.environ:
            settings[key.lower()] = os.environ[key]

    return settings'''

new = '''    settings = {}

    # 1. baked-in defaults first, so a bare container still boots.
    #    These are the lowest-precedence source.
    for key, value in _load_env_file(os.path.join(CONFIG_DIR, "defaults.env")).items():
        settings[key.lower()] = value

    # 2. environment-specific yaml overrides the defaults
    settings.update(_parse_yaml(os.path.join(CONFIG_DIR, f"{env}.yaml")))

    # 3. real environment variables win over everything
    for key in ("DATABASE_URL", "REDIS_URL", "LOG_LEVEL"):
        if key in os.environ:
            settings[key.lower()] = os.environ[key]

    return settings'''

assert old in src, "expected block not found"
p.write_text(src.replace(old, new))
